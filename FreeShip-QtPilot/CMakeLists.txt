cmake_minimum_required(VERSION 3.24)
project(FreeShipQtPilot VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets OpenGL)
add_library(linesplan src/scene.cpp src/view.cpp src/gpu_fill.cpp)
target_include_directories(linesplan PUBLIC src)
target_link_libraries(linesplan PUBLIC Qt6::Widgets Qt6::OpenGL)
target_compile_options(linesplan PRIVATE -Wall -Wextra)
add_executable(FreeShipQtPilot WIN32 src/main.cpp)
target_link_libraries(FreeShipQtPilot PRIVATE linesplan)
set(PILOT_TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../FreeShip-Tests/qt/linesplan_tests.cpp" CACHE FILEPATH "Companion test entry")
add_executable(qt_lines_tests "${PILOT_TEST_SOURCE}")
target_link_libraries(qt_lines_tests PRIVATE linesplan)
